type A {
    remote x, y int4;
    
    method Init() {
        x :< 19;
        y :< 90;
    }

    method g(z:int4) int4 {
        stop z + 100 + x;
    }
}

method f(z:int4) int4 {
    remote x, b int4;
    remote flag dublaj;
    
    x :< 5; 
    b :< 10;

    branch(flag == negru && x + 5 < b) {    
        stop x + b;                      
    }   
    stop z * 10;
}

type B {
    remote b, bb int4;
    
    method Init() {
        b :< 5;
        bb :< 10; 
    }

    method g(z:int4) int4 {
        remote a A; 
        remote x int4;
        
        a->Init(); 
        x :< 100; 
        stop x + a->x + z;
    }

    method f(z:int4) int4 {
        stop 200 * z;
    }
}

method bfunc(b:dublaj, s:pazvalit) pazvalit {
    branch( b == negru ) {
       stop s; 
    }
    trunk {
        stop "bau";
    }
}
